def evaluate(y_true, y_pred):
    mae = mean_absolute_error(y_true, y_pred)

    rmse = np.sqrt(
        mean_squared_error(y_true, y_pred)
    )

    r2 = r2_score(y_true, y_pred)

    return mae, rmse, r2


def make_models():
    rf_model = RandomForestRegressor(
        n_estimators=500,
        random_state=42,
        n_jobs=-1,
        max_depth=None,
        min_samples_leaf=2
    )

    xgb_model = XGBRegressor(
        n_estimators=500,
        learning_rate=0.03,
        max_depth=4,
        subsample=0.8,
        colsample_bytree=0.8,
        random_state=42,
        objective="reg:squarederror",
        n_jobs=-1
    )

    return rf_model, xgb_model


def preprocess_data(data, features):
    use_cols = features + [TARGET, "base_quarter"]

    temp = data[use_cols].copy()
    temp = temp.dropna()

    categorical_cols = []

    for col in features:
        if temp[col].dtype == "object":
            categorical_cols.append(col)

    if len(categorical_cols) > 0:
        temp = pd.get_dummies(
            temp,
            columns=categorical_cols,
            drop_first=True
        )

    return temp


def get_feature_cols(data):
    return [
        col for col in data.columns
        if col not in [TARGET, "base_quarter"]
    ]


def run_single_train_test(data, feature_cols):
    train_df = data[
        data["base_quarter"] < TEST_QUARTER
    ].copy()

    test_df = data[
        data["base_quarter"] == TEST_QUARTER
    ].copy()

    if len(train_df) == 0 or len(test_df) == 0:
        return None, None, None, None, None

    X_train = train_df[feature_cols]
    y_train = train_df[TARGET]

    X_test = test_df[feature_cols]
    y_test = test_df[TARGET]

    rf_model, xgb_model = make_models()

    rf_model.fit(X_train, y_train)
    xgb_model.fit(X_train, y_train)

    rf_pred = rf_model.predict(X_test)
    xgb_pred = xgb_model.predict(X_test)

    rf_mae, rf_rmse, rf_r2 = evaluate(y_test, rf_pred)
    xgb_mae, xgb_rmse, xgb_r2 = evaluate(y_test, xgb_pred)

    result_df = pd.DataFrame({
        "Model": ["Random Forest", "XGBoost"],
        "MAE": [rf_mae, xgb_mae],
        "RMSE": [rf_rmse, xgb_rmse],
        "R2": [rf_r2, xgb_r2],
        "Train Rows": [len(train_df), len(train_df)],
        "Test Rows": [len(test_df), len(test_df)],
        "Test Quarter": [TEST_QUARTER, TEST_QUARTER],
    })

    rf_importance = pd.DataFrame({
        "feature": feature_cols,
        "importance": rf_model.feature_importances_
    }).sort_values("importance", ascending=False)

    xgb_importance = pd.DataFrame({
        "feature": feature_cols,
        "importance": xgb_model.feature_importances_
    }).sort_values("importance", ascending=False)

    pred_df = pd.DataFrame({
        "test_quarter": TEST_QUARTER,
        "actual": y_test.values,
        "rf_pred": rf_pred,
        "xgb_pred": xgb_pred,
    })

    return result_df, rf_importance, xgb_importance, pred_df, train_df, test_df


def run_rolling_validation(data, feature_cols, min_train_quarters=4):
    quarters = sorted(data["base_quarter"].unique())

    results = []
    all_preds = []

    last_rf_importance = None
    last_xgb_importance = None

    for test_quarter in quarters[min_train_quarters:]:

        train_df = data[
            data["base_quarter"] < test_quarter
        ].copy()

        test_df = data[
            data["base_quarter"] == test_quarter
        ].copy()

        if len(train_df) == 0 or len(test_df) == 0:
            continue

        X_train = train_df[feature_cols]
        y_train = train_df[TARGET]

        X_test = test_df[feature_cols]
        y_test = test_df[TARGET]

        rf_model, xgb_model = make_models()

        rf_model.fit(X_train, y_train)
        xgb_model.fit(X_train, y_train)

        rf_pred = rf_model.predict(X_test)
        xgb_pred = xgb_model.predict(X_test)

        rf_mae, rf_rmse, rf_r2 = evaluate(y_test, rf_pred)
        xgb_mae, xgb_rmse, xgb_r2 = evaluate(y_test, xgb_pred)

        results.append({
            "test_quarter": test_quarter,
            "train_start": train_df["base_quarter"].min(),
            "train_end": train_df["base_quarter"].max(),
            "train_rows": len(train_df),
            "test_rows": len(test_df),

            "RF_MAE": rf_mae,
            "RF_RMSE": rf_rmse,
            "RF_R2": rf_r2,

            "XGB_MAE": xgb_mae,
            "XGB_RMSE": xgb_rmse,
            "XGB_R2": xgb_r2,
        })

        temp_pred = pd.DataFrame({
            "test_quarter": test_quarter,
            "actual": y_test.values,
            "rf_pred": rf_pred,
            "xgb_pred": xgb_pred,
        })

        all_preds.append(temp_pred)

        last_rf_importance = pd.DataFrame({
            "feature": feature_cols,
            "importance": rf_model.feature_importances_
        }).sort_values("importance", ascending=False)

        last_xgb_importance = pd.DataFrame({
            "feature": feature_cols,

def evaluate(y_true, y_pred):
    mae = mean_absolute_error(y_true, y_pred)

    rmse = np.sqrt(
        mean_squared_error(y_true, y_pred)
    )

    r2 = r2_score(y_true, y_pred)

    return mae, rmse, r2


def make_models():
    rf_model = RandomForestRegressor(
        n_estimators=500,
        random_state=42,
        n_jobs=-1,
        max_depth=None,
        min_samples_leaf=2
    )

    xgb_model = XGBRegressor(
        n_estimators=500,
        learning_rate=0.03,
        max_depth=4,
        subsample=0.8,
        colsample_bytree=0.8,
        random_state=42,
        objective="reg:squarederror",
        n_jobs=-1
    )

    return rf_model, xgb_model


def preprocess_data(data, features):
    use_cols = list(set(
    features + [TARGET, "base_quarter"] + META_COLUMNS
))

    temp = data[use_cols].copy()
    temp = temp.dropna()

    categorical_cols = []

    for col in features:
        if temp[col].dtype == "object":
            categorical_cols.append(col)

    if len(categorical_cols) > 0:
        temp = pd.get_dummies(
            temp,
            columns=categorical_cols,
            drop_first=True
        )

    return temp


def get_feature_cols(data):
    exclude_cols = [
        TARGET,
        "base_quarter"
    ] + META_COLUMNS

    return [
        col for col in data.columns
        if col not in exclude_cols
    ]


def run_single_train_test(data, feature_cols):
    train_df = data[
        data["base_quarter"] < TEST_QUARTER
    ].copy()

    test_df = data[
        data["base_quarter"] == TEST_QUARTER
    ].copy()

    if len(train_df) == 0 or len(test_df) == 0:
        return None, None, None, None, None

    X_train = train_df[feature_cols]
    y_train = train_df[TARGET]

    X_test = test_df[feature_cols]
    y_test = test_df[TARGET]

    rf_model, xgb_model = make_models()

    rf_model.fit(X_train, y_train)
    xgb_model.fit(X_train, y_train)

    rf_pred = rf_model.predict(X_test)
    xgb_pred = xgb_model.predict(X_test)

    rf_mae, rf_rmse, rf_r2 = evaluate(y_test, rf_pred)
    xgb_mae, xgb_rmse, xgb_r2 = evaluate(y_test, xgb_pred)

    result_df = pd.DataFrame({
        "Model": ["Random Forest", "XGBoost"],
        "MAE": [rf_mae, xgb_mae],
        "RMSE": [rf_rmse, xgb_rmse],
        "R2": [rf_r2, xgb_r2],
        "Train Rows": [len(train_df), len(train_df)],
        "Test Rows": [len(test_df), len(test_df)],
        "Test Quarter": [TEST_QUARTER, TEST_QUARTER],
    })

    rf_importance = pd.DataFrame({
        "feature": feature_cols,
        "importance": rf_model.feature_importances_
    }).sort_values("importance", ascending=False)

    xgb_importance = pd.DataFrame({
        "feature": feature_cols,
        "importance": xgb_model.feature_importances_
    }).sort_values("importance", ascending=False)

    pred_df = pd.DataFrame({
    "test_quarter": TEST_QUARTER,

    "adm_nm": test_df["adm_nm"].values,
    "신구축여부": test_df["신구축여부"].values,
    "전용면적그룹": test_df["전용면적그룹"].values,

    "actual": y_test.values,
    "rf_pred": rf_pred,
    "xgb_pred": xgb_pred,
})

    return result_df, rf_importance, xgb_importance, pred_df, train_df, test_df


def run_rolling_validation(data, feature_cols, min_train_quarters=4):
    quarters = sorted(data["base_quarter"].unique())

    results = []
    all_preds = []

    last_rf_importance = None
    last_xgb_importance = None

    for test_quarter in quarters[min_train_quarters:]:

        train_df = data[
            data["base_quarter"] < test_quarter
        ].copy()

        test_df = data[
            data["base_quarter"] == test_quarter
        ].copy()

        if len(train_df) == 0 or len(test_df) == 0:
            continue

        X_train = train_df[feature_cols]
        y_train = train_df[TARGET]

        X_test = test_df[feature_cols]
        y_test = test_df[TARGET]

        rf_model, xgb_model = make_models()

        rf_model.fit(X_train, y_train)
        xgb_model.fit(X_train, y_train)

        rf_pred = rf_model.predict(X_test)
        xgb_pred = xgb_model.predict(X_test)

        rf_mae, rf_rmse, rf_r2 = evaluate(y_test, rf_pred)
        xgb_mae, xgb_rmse, xgb_r2 = evaluate(y_test, xgb_pred)

        results.append({
            "test_quarter": test_quarter,
            "train_start": train_df["base_quarter"].min(),
            "train_end": train_df["base_quarter"].max(),
            "train_rows": len(train_df),
            "test_rows": len(test_df),

            "RF_MAE": rf_mae,
            "RF_RMSE": rf_rmse,
            "RF_R2": rf_r2,

            "XGB_MAE": xgb_mae,
            "XGB_RMSE": xgb_rmse,
            "XGB_R2": xgb_r2,
        })

        temp_pred = pd.DataFrame({
            "test_quarter": test_quarter,
            "adm_nm": test_df["adm_nm"].values,
            "신구축여부": test_df["신구축여부"].values,
            "전용면적그룹": test_df["전용면적그룹"].values,
            
            "actual": y_test.values,
            "rf_pred": rf_pred,
            "xgb_pred": xgb_pred,})

        all_preds.append(temp_pred)

        last_rf_importance = pd.DataFrame({
            "feature": feature_cols,
            "importance": rf_model.feature_importances_
        }).sort_values("importance", ascending=False)

        last_xgb_importance = pd.DataFrame({
            "feature": feature_cols,
            "importance": xgb_model.feature_importances_
        }).sort_values("importance", ascending=False)

    result_df = pd.DataFrame(results)

    if len(all_preds) > 0:
        pred_df = pd.concat(all_preds, ignore_index=True)
    else:
        pred_df = pd.DataFrame()

    return result_df, pred_df, last_rf_importance, last_xgb_importance


def add_feature_group(title, feature_list, default=True):
    st.sidebar.subheader(title)

    selected = []

    for feature in feature_list:
        if feature not in df.columns:
            continue

        checked = st.sidebar.checkbox(
            feature,
            value=default,
            key=feature
        )

        if checked:
            selected.append(feature)

    return selected

def read_csv_auto(path):
    for enc in ["utf-8-sig", "cp949", "euc-kr", "utf-8"]:
        try:
            return pd.read_csv(path, encoding=enc, low_memory=False)
        except UnicodeDecodeError:
            continue

    raise ValueError(f"인코딩 실패: {path}")


def clean_text(text):
    if not isinstance(text, str):
        return ""

    text = re.sub(r"\s+", " ", text)
    return text.strip()


def is_empty_url(url):
    return pd.isna(url) or str(url).strip() == ""


def should_save_to_jsonl(record):
    """
    checkpoint jsonl에 저장할지 판단.
    empty_url은 저장하지 않음.
    """
    if not record:
        return False

    return record.get("원문추출상태") != EMPTY_URL_STATUS


def append_jsonl(records, path):
    """
    checkpoint jsonl에 append.
    empty_url은 저장하지 않음.
    """
    records = [
        record for record in records
        if should_save_to_jsonl(record)
    ]

    if not records:
        return

    with open(path, "a", encoding="utf-8") as f:
        for record in records:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")


def load_checkpoint(path):
    if not path.exists():
        return pd.DataFrame()

    rows = []

    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            try:
                rows.append(json.loads(line))
            except Exception:
                continue

    if not rows:
        return pd.DataFrame()

    checkpoint_df = pd.DataFrame(rows)

    if "index" not in checkpoint_df.columns:
        return pd.DataFrame()

    checkpoint_df = checkpoint_df.drop_duplicates(
        subset=["index"],
        keep="last"
    )

    return checkpoint_df


def load_success_urls(path):
    """
    이미 성공한 URL 목록 불러오기.
    기존 수집 결과를 기준으로 재수집 방지.
    """
    if not path.exists():
        return set()

    urls = set()

    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            try:
                row = json.loads(line)
            except Exception:
                continue

            status = row.get("원문추출상태", "")
            url = str(row.get("url", "")).strip()

            if status == SUCCESS_STATUS and url:
                urls.add(url)

    return urls


def rewrite_checkpoint_success_only(path):
    """
    재시작 시 checkpoint 정리.

    - success만 남김
    - failed / timeout_skipped / too_short / empty_url 등은 제거
    - 따라서 실패했던 index는 다시 요청됨
    - empty_url은 그냥 삭제되고, 이후에도 jsonl에 저장되지 않음
    """
    checkpoint_df = load_checkpoint(path)

    if checkpoint_df.empty:
        path.write_text("", encoding="utf-8")
        return pd.DataFrame()

    checkpoint_df = checkpoint_df.drop_duplicates(
        subset=["index"],
        keep="last"
    )

    success_df = checkpoint_df[
        checkpoint_df["원문추출상태"] == SUCCESS_STATUS
    ].copy()

    with open(path, "w", encoding="utf-8") as f:
        for record in success_df.to_dict("records"):
            f.write(json.dumps(record, ensure_ascii=False) + "\n")

    return success_df

def fetch_html(url):
    response = requests.get(
        url,
        headers=HEADERS,
        timeout=(5, TIMEOUT)
    )
    response.raise_for_status()

    response.encoding = response.apparent_encoding
    return response.text

def extract_by_trafilatura(html, url):
    text = trafilatura.extract(
        html,
        url=url,
        include_comments=False,
        include_tables=False,
        favor_recall=True
    )
    return clean_text(text)


# =========================
# 추출 방식 2: readability
# =========================

def extract_by_readability(html):
    doc = Document(html)
    summary_html = doc.summary()

    soup = BeautifulSoup(summary_html, "lxml")
    text = soup.get_text(" ", strip=True)

    return clean_text(text)


# =========================
# 추출 방식 3: BeautifulSoup
# =========================

def extract_by_bs4(html):
    soup = BeautifulSoup(html, "lxml")

    for tag in soup(["script", "style", "noscript", "iframe"]):
        tag.decompose()

    selectors = [
        "article",
        "#articleBody",
        "#article_body",
        "#articeBody",
        "#news_body_area",
        "#newsView",
        ".article_view",
        ".article-body",
        ".article_body",
        ".news_body",
        ".news-view",
        ".view_con",
        ".view_content",
        ".article_txt",
        ".art_txt",
        ".read_txt",
        ".news_cnt_detail_wrap",
    ]

    candidates = []

    for selector in selectors:
        for elem in soup.select(selector):
            text = clean_text(elem.get_text(" ", strip=True))
            if text:
                candidates.append(text)

    if candidates:
        return max(candidates, key=len)

    paragraphs = [
        clean_text(p.get_text(" ", strip=True))
        for p in soup.find_all("p")
    ]
    paragraphs = [p for p in paragraphs if len(p) >= 30]

    return clean_text(" ".join(paragraphs))

def process_one(row_dict):
    idx = row_dict["index"]
    url = row_dict.get(URL_COL, "")
    original_body = row_dict.get(ORIGINAL_BODY_COL, "")

    if is_empty_url(url):
        return {
            "index": idx,
            FULL_TEXT_COL: original_body,
            "원문추출방법": "original_body",
            "원문추출상태": EMPTY_URL_STATUS,
            "fail_reason": "empty_url",
            "url": url,
        }

    url = str(url).strip()

    try:
        if SLEEP_SEC > 0:
            time.sleep(SLEEP_SEC)

        html = fetch_html(url)

        candidates = []

        extractors = [
            ("trafilatura", lambda: extract_by_trafilatura(html, url)),
            ("readability", lambda: extract_by_readability(html)),
            ("bs4", lambda: extract_by_bs4(html)),
        ]

        for method_name, extractor in extractors:
            try:
                text = extractor()
                if text:
                    candidates.append((method_name, text))
            except Exception:
                pass

        if not candidates:
            return {
                "index": idx,
                FULL_TEXT_COL: original_body,
                "원문추출방법": "original_body",
                "원문추출상태": "failed",
                "fail_reason": "no_text_extracted",
                "url": url,
            }

        best_method, best_text = max(candidates, key=lambda x: len(x[1]))

        if len(best_text) < MIN_TEXT_LEN:
            return {
                "index": idx,
                FULL_TEXT_COL: original_body,
                "원문추출방법": "original_body",
                "원문추출상태": "too_short",
                "fail_reason": f"too_short_{len(best_text)}",
                "url": url,
            }

        return {
            "index": idx,
            FULL_TEXT_COL: best_text,
            "원문추출방법": best_method,
            "원문추출상태": SUCCESS_STATUS,
            "fail_reason": "",
            "url": url,
        }

    except Exception as e:
        return {
            "index": idx,
            FULL_TEXT_COL: original_body,
            "원문추출방법": "original_body",
            "원문추출상태": "failed",
            "fail_reason": str(e),
            "url": url,
        }

    if checkpoint_df.empty:
        path.write_text("", encoding="utf-8")
        return pd.DataFrame()

    checkpoint_df = checkpoint_df.drop_duplicates(
        subset=["index"],
        keep="last"
    )

    success_df = checkpoint_df[
        checkpoint_df["원문추출상태"] == SUCCESS_STATUS
    ].copy()

            f.write(json.dumps(record, ensure_ascii=False) + "\n")

# =========================
# HTML 요청
# =========================

def fetch_html(url):
    """
    429는 오래 기다리지 않고 바로 failed 처리.
    500번대 서버 오류만 짧게 재시도.
    """
    last_error = None

    for attempt in range(1, 4):
        try:
            response = requests.get(
                url,
                headers=HEADERS,
                timeout=(5, TIMEOUT)
            )

            # 429는 Retry-After가 길 수 있으므로 바로 실패 처리
            if response.status_code == 429:
                raise requests.HTTPError("429 Too Many Requests")

            # 서버 오류는 짧게만 재시도
            if response.status_code in [500, 502, 503, 504]:
                wait_sec = 5 * attempt
                print(f"[서버 오류 재시도] status={response.status_code}, {wait_sec}초 대기: {url}")
                time.sleep(wait_sec)
                last_error = requests.HTTPError(f"{response.status_code} Server Error")
                continue

            response.raise_for_status()
            response.encoding = response.apparent_encoding
            return response.text

        except requests.HTTPError as e:
            if "429" in str(e):
                raise e

            last_error = e
            time.sleep(3 * attempt)

        except Exception as e:
            last_error = e
            time.sleep(3 * attempt)

    raise last_error

def extract_by_trafilatura(html, url):
    text = trafilatura.extract(
        html,
        url=url,
        include_comments=False,
        include_tables=False,
        favor_recall=True
    )

def extract_by_readability(html):
    doc = Document(html)
    summary_html = doc.summary()

    soup = BeautifulSoup(summary_html, "lxml")
    text = soup.get_text(" ", strip=True)

    return clean_text(text)

# 추출 방식 3: BeautifulSoup
# =========================

def extract_by_bs4(html):
    soup = BeautifulSoup(html, "lxml")

    for tag in soup(["script", "style", "noscript", "iframe"]):
        tag.decompose()

    selectors = [
        "article",
        "#articleBody",
        "#article_body",
        "#articeBody",
        "#news_body_area",
        "#newsView",
        ".article_view",
        ".article-body",
        ".article_body",
        ".news_body",
        ".news-view",
        ".view_con",
        ".view_content",
        ".article_txt",
        ".art_txt",
        ".read_txt",
        ".news_cnt_detail_wrap",
    ]

    candidates = []

    for selector in selectors:
        for elem in soup.select(selector):
            text = clean_text(elem.get_text(" ", strip=True))
            if text:
                candidates.append(text)

    if candidates:
        return max(candidates, key=len)

    paragraphs = [
        clean_text(p.get_text(" ", strip=True))
        for p in soup.find_all("p")
    ]
    paragraphs = [p for p in paragraphs if len(p) >= 30]

    return clean_text(" ".join(paragraphs))

def process_one(row_dict):
    idx = row_dict["index"]
    url = row_dict.get(URL_COL, "")
    original_body = row_dict.get(ORIGINAL_BODY_COL, "")

    if is_empty_url(url):
        return {
            "index": idx,
            FULL_TEXT_COL: original_body,
            "원문추출방법": "original_body",
            "원문추출상태": EMPTY_URL_STATUS,
            "fail_reason": "empty_url",
            "url": url,
        }

    url = str(url).strip()

    try:
        if SLEEP_SEC > 0:
            time.sleep(SLEEP_SEC)

        html = fetch_html(url)

        candidates = []

        extractors = [
            ("trafilatura", lambda: extract_by_trafilatura(html, url)),
            ("readability", lambda: extract_by_readability(html)),
            ("bs4", lambda: extract_by_bs4(html)),
        ]

        for method_name, extractor in extractors:
            try:
                text = extractor()
                if text:
                    candidates.append((method_name, text))
            except Exception:
                pass

        if not candidates:
            return {
                "index": idx,
                FULL_TEXT_COL: original_body,
                "원문추출방법": "original_body",
                "원문추출상태": "failed",
                "fail_reason": "no_text_extracted",
                "url": url,
            }

        best_method, best_text = max(candidates, key=lambda x: len(x[1]))

        if len(best_text) < MIN_TEXT_LEN:
            return {
                "index": idx,
                FULL_TEXT_COL: original_body,
                "원문추출방법": "original_body",
                "원문추출상태": "too_short",
                "fail_reason": f"too_short_{len(best_text)}",
                "url": url,
            }

        return {
            "index": idx,
            FULL_TEXT_COL: best_text,
            "원문추출방법": best_method,
            "원문추출상태": SUCCESS_STATUS,
            "fail_reason": "",
            "url": url,
        }

    except Exception as e:
        return {
            "index": idx,
            FULL_TEXT_COL: original_body,
            "원문추출방법": "original_body",
            "원문추출상태": "failed",
            "fail_reason": str(e),
            "url": url,
        }
